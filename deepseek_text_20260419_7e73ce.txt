function App() {
  const [isNative, setIsNative] = useState(false)
  const [permissionsGranted, setPermissionsGranted] = useState(true)
  const [initialized, setInitialized] = useState(false)
  const [offlineContent, setOfflineContent] = useState([])
  const [showTownSelector, setShowTownSelector] = useState(false)

  useEffect(() => {
    const init = async () => {
      const native = Capacitor.isNativePlatform()
      setIsNative(native)
      
      if (native) {
        await AndroidBridge.initialize()
        const granted = await AndroidBridge.requestPermissions()
        setPermissionsGranted(granted)
        
        const offline = await AndroidBridge.getOfflineContentList()
        setOfflineContent(offline || [])
      }
      
      setInitialized(true)
    }
    
    init()
  }, [])

  if (!initialized) {
    return (
      <div className="splash-screen">
        <div className="splash-content">
          <img src="/ellipses-icon.svg" alt="Ellipses" />
          <h1 className="app-brand">{APP_NAME}</h1>
          <h2 className="welcome-header">Welcome To...</h2>
          <h3 className="town-name">{CURRENT_TOWN}</h3>
          <div className="loading-spinner"></div>
          <p className="splash-subtitle">Discover the stories between the lines</p>
        </div>
      </div>
    )
  }

  if (!permissionsGranted) {
    return (
      <div className="permission-denied">
        <Camera size={64} color="#e74c3c" />
        <h2>Permissions Required</h2>

        <button onClick={() => window.location.reload()} className="grant-button">
          Grant Permissions
        </button>

  return (

          <div className="header-left">
            <Link to="/" className="brand-link">
              <img src="/ellipses-icon.svg" alt="Ellipses" className="header-icon" />

import { useEffect, useRef, useState } from 'react'
import * as LocAR from 'locar'
import { supabase } from '../lib/supabaseClient'
import { AndroidBridge } from '../lib/androidBridge'
import { haversineDistance, formatDistance, getCardinalDirection } from '../lib/geospatial'
import { X, BookOpen, MapPin, Volume2, Share2, Download, Navigation, Sparkles } from 'lucide-react'

const CATEGORY_COLORS = {
  industrial_mill: '#e67e22',
  cook_historic_site: '#2980b9',
  railway_narrow_gauge: '#8e44ad',
  garden_city_architecture: '#27ae60',
  civic_building: '#c0392b',
  ecclesiastical: '#f1c40f',
  indigenous_context: '#1abc9c',
  port_shipping: '#34495e'
}

const CATEGORY_ICONS = {
  industrial_mill: '🏭',
  cook_historic_site: '🗺️',
  railway_narrow_gauge: '🚂',
  garden_city_architecture: '🏛️',
  civic_building: '🏫',
  ecclesiastical: '⛪',
  indigenous_context: '🌿',
  port_shipping: '⚓'
}

export default function ARView({ appName, townName, isNative }) {
  const sceneRef = useRef(null)
  const [assets, setAssets] = useState([])
  const [selectedAsset, setSelectedAsset] = useState(null)
  const [userLocation, setUserLocation] = useState(null)
  const [compassHeading, setCompassHeading] = useState(0)
  const [isOffline, setIsOffline] = useState(false)
  const [arReady, setArReady] = useState(false)
  const [nearbyCount, setNearbyCount] = useState(0)

  useEffect(() => {
    if (isNative) {
      AndroidBridge.checkNetworkStatus().then(connected => setIsOffline(!connected))
      AndroidBridge.onNetworkChange(status => setIsOffline(!status.connected))
    }

    if (window.DeviceOrientationEvent) {
      window.addEventListener('deviceorientation', handleOrientation)
    }
    return () => window.removeEventListener('deviceorientation', handleOrientation)
  }, [isNative])

  const handleOrientation = (event) => {
    if (event.webkitCompassHeading) {
      setCompassHeading(event.webkitCompassHeading)
    } else if (event.alpha) {
      setCompassHeading(360 - event.alpha)
    }
  }

  useEffect(() => {
    if (!userLocation) return

    const fetchNearby = async () => {
      try {
        let data
        if (isOffline && isNative) {
          const offlineContent = await AndroidBridge.getOfflineContentList()
          data = offlineContent.filter(asset => {
            if (!asset.location?.coordinates) return false
            const [lng, lat] = asset.location.coordinates
            return haversineDistance(userLocation.latitude, userLocation.longitude, lat, lng) <= 1000
          })
        } else {
          const { data: fetchedData } = await supabase.rpc('get_nearby_assets', {
            lat: userLocation.latitude,
            lng: userLocation.longitude,
            radius_meters: 1000
          })
          data = fetchedData || []
        }
        setAssets(data)
        setNearbyCount(data.length)
      } catch (error) {
        console.error('Error fetching nearby assets:', error)
      }
    }

    fetchNearby()
  }, [userLocation, isOffline, isNative])

  useEffect(() => {
    if (!sceneRef.current) return

    const scene = new LocAR.Scene({
      camera: { gpsMinDistance: 5, gpsTimeInterval: 3000 }
    })

    const camera = new LocAR.Camera()
    scene.add(camera)
    sceneRef.current.appendChild(scene.domElement)

    const watchId = navigator.geolocation.watchPosition(
      (pos) => {
        setUserLocation({
          latitude: pos.coords.latitude,
          longitude: pos.coords.longitude,
          altitude: pos.coords.altitude || 0,
          accuracy: pos.coords.accuracy
        })
        setArReady(true)
      },
      (err) => console.error('Geolocation error:', err),
      { enableHighAccuracy: true, maximumAge: 0, timeout: 5000 }
    )

    return () => {
      navigator.geolocation.clearWatch(watchId)
      scene.dispose()
    }
  }, [])

  useEffect(() => {
    if (!sceneRef.current || !assets.length) return
    const scene = sceneRef.current.scene
    
    while (scene.firstChild) scene.removeChild(scene.firstChild)

    assets.forEach(asset => {
      if (!asset.location?.coordinates) return
      
      const [lng, lat] = asset.location.coordinates
      const distance = userLocation ? haversineDistance(
        userLocation.latitude, userLocation.longitude, lat, lng
      ) : 0
      
      const entity = document.createElement('a-entity')
      entity.setAttribute('gps-new-entity-place', { latitude: lat, longitude: lng })

      const color = CATEGORY_COLORS[asset.category] || '#c0392b'
      const icon = CATEGORY_ICONS[asset.category] || '📍'
      
      entity.setAttribute('geometry', { primitive: 'cylinder', height: 3 + (distance / 100), radius: 0.8 })
      entity.setAttribute('material', { color: color, transparent: true, opacity: 0.85 })

  indigenous_context: '#1abc9c',
  port_shipping: '#34495e'

export default function MapView({ appName, townName, isNative }) {
  const mapContainer = useRef(null)
  const map = useRef(null)
  const userMarker = useRef(null)
  const [assets, setAssets] = useState([])
  const [userLocation, setUserLocation] = useState(null)
  const [mapLoaded, setMapLoaded] = useState(false)

  useEffect(() => {
    if (!mapContainer.current || map.current) return

    map.current = new maplibregl.Map({
      container: mapContainer.current,
      style: `${MAP_STYLE}?key=${import.meta.env.VITE_MAPLIBRE_TOKEN}`,
      center: [-57.9519, 48.9532],
      zoom: 13
    })

    map.current.addControl(new maplibregl.NavigationControl(), 'top-right')
    map.current.addControl(new maplibregl.GeolocateControl({
      positionOptions: { enableHighAccuracy: true },
      trackUserLocation: true,
      showUserHeading: true
    }))

    map.current.on('load', () => setMapLoaded(true))

    const watchId = navigator.geolocation.watchPosition(
      (pos) => {
        const location = {
          latitude: pos.coords.latitude,
          longitude: pos.coords.longitude
        }
        setUserLocation(location)
        
        if (map.current && !userMarker.current) {
          userMarker.current = new maplibregl.Marker({ color: '#1a1a2e' })
            .setLngLat([location.longitude, location.latitude])
            .setPopup(new maplibregl.Popup().setText('You are here'))
            .addTo(map.current)
          
          map.current.flyTo({ 
            center: [location.longitude, location.latitude], 
            zoom: 15 
          })
        } else if (userMarker.current) {
          userMarker.current.setLngLat([location.longitude, location.latitude])
        }
      },
      (err) => console.error('Geolocation error:', err),
      { enableHighAccuracy: true }
    )

    return () => {
      navigator.geolocation.clearWatch(watchId)
      map.current?.remove()
    }
  }, [])

    if (!mapLoaded || !userLocation) return

      try {

          lat: userLocation.latitude,
          lng: userLocation.longitude,

          </div>
        </div>
      `)

        .setLngLat([lng, lat])
        .setPopup(popup)
        .addTo(map.current)

  })

      .channel('heritage_assets_changes')
      .on('postgres_changes', { event: '*', schema: 'public', table: 'heritage_assets' }, fetchAllData)
      .subscribe()

    return () => subscription.unsubscribe()
  }, [])

    await Promise.all([fetchPending(), fetchApproved()])
    setLoading(false)

      .from('heritage_assets')
      .select('*')
      .eq('status', 'pending')
      .order('created_at', { ascending: false })
    
    setPendingStories(data || [])
  }

  const fetchApproved = async () => {
    const { data } = await supabase
      .from('heritage_assets')
      .select('*')
      .eq('status', 'approved')
      .order('created_at', { ascending: false })
    
    setApprovedStories(data || [])
    
    const categoryCount = {}
    const contributors = new Set()
    
    data?.forEach(story => {
      categoryCount[story.category] = (categoryCount[story.category] || 0) + 1
      if (story.contributor) contributors.add(story.contributor)
    })
    
    setMetrics({
      totalApproved: data?.length || 0,
      totalPending: pendingStories.length,
      totalContributors: contributors.size,
      categories: categoryCount
    })
  }

    await supabase
      .from('heritage_assets')
      .update({ status, updated_at: new Date().toISOString() })
      .eq('id', id)
    
    fetchAllData()

      year: 'numeric', month: 'long', day: 'numeric'
    })
  }

        </div>
      </div>

import { Network } from '@capacitor/network'
import { WifiOff, Database } from 'lucide-react'

export default function OfflineNotice({ offlineCount = 0 }) {
  const [isOffline, setIsOffline] = useState(false)

  useEffect(() => {
    const checkNetwork = async () => {
      const status = await Network.getStatus()
      setIsOffline(!status.connected)

      })
    }
    
    checkNetwork()

    <div className="offline-banner">
      <WifiOff size={16} />
      <span>Offline Mode</span>
      {offlineCount > 0 && (
        <>
          <Database size={14} />
          <span>{offlineCount} stories cached</span>
        </>
      )}
    </div>
  )
}

import { Capacitor } from '@capacitor/core'

import { Network } from '@capacitor/network'

export class AndroidBridge {
  static async initialize() {
    if (!Capacitor.isNativePlatform()) return
    
    try {
      await SplashScreen.hide()
      await StatusBar.setStyle({ style: 'DARK' })
      await StatusBar.setBackgroundColor({ color: '#1a1a2e' })
      
      const info = await Device.getInfo()
      console.log(`Ellipses running on: ${info.platform} ${info.osVersion}`)
      
      App.addListener('backButton', ({ canGoBack }) => {
        if (canGoBack) window.history.back()
        else App.minimizeApp()
      })
    } catch (error) {
      console.error('Android bridge initialization error:', error)
    }
  }
  
  static async requestPermissions() {
    if (!Capacitor.isNativePlatform()) return true
    
    try {
      const locationPerm = await Geolocation.requestPermissions()
      const cameraPerm = await Camera.requestPermissions()
      return locationPerm.location === 'granted' && cameraPerm.camera === 'granted'
    } catch (error) {
      console.error('Permission request error:', error)
      return false
    }
  }
  
  static async shareStory(story) {
    try {
      const shareData = {
        title: `${story.title} — Ellipses`,
        text: `${story.short_description}\n\nDiscover more stories with Ellipses: Welcome To... ${story.townName || ''}`,
        url: `https://ellipses.app/story/${story.id}`,
        dialogTitle: 'Share via Ellipses'
      }
      
      if (Capacitor.isNativePlatform()) {
        await Share.share(shareData)
      } else if (navigator.share) {
        await navigator.share(shareData)
      } else {
        await navigator.clipboard.writeText(shareData.url)
        alert('Link copied to clipboard!')
      }
    } catch (error) {
      console.error('Share error:', error)
    }
  }
  
  static async saveOfflineContent(storyId, content) {
    if (!Capacitor.isNativePlatform()) return false
    
    try {
      const fileName = `story_${storyId}.json`
      await Filesystem.writeFile({
        path: `offline/${fileName}`,
        data: JSON.stringify({ ...content, cached_at: new Date().toISOString() }),
        directory: Directory.Data,
        encoding: 'utf8'
      })
      
      const index = await this.getOfflineIndex()
      if (!index.includes(storyId)) {
        index.push(storyId)
        await Filesystem.writeFile({
          path: 'offline/index.json',
          data: JSON.stringify(index),
          directory: Directory.Data,
          encoding: 'utf8'
        })
      }
      
      return true
    } catch (error) {
      console.error('Failed to save offline content:', error)
      return false
    }
  }
  
  static async getOfflineContent(storyId) {
    if (!Capacitor.isNativePlatform()) return null
    
    try {
      const result = await Filesystem.readFile({
        path: `offline/story_${storyId}.json`,
        directory: Directory.Data,
        encoding: 'utf8'
      })
      return JSON.parse(result.data)
    } catch {
      return null
    }
  }
  
  static async getOfflineIndex() {
    if (!Capacitor.isNativePlatform()) return []
    
    try {
      const result = await Filesystem.readFile({
        path: 'offline/index.json',
        directory: Directory.Data,
        encoding: 'utf8'
      })
      return JSON.parse(result.data)
    } catch {
      return []
    }
  }
  
  static async getOfflineContentList() {
    const index = await this.getOfflineIndex()
    const contents = []
    
    for (const id of index) {
      const content = await this.getOfflineContent(id)
      if (content) contents.push(content)
    }
    
    return contents
  }
  
  static async checkNetworkStatus() {
    try {
      const status = await Network.getStatus()
      return status.connected
    } catch {
      return true
    }
  }
  
  static onNetworkChange(callback) {
    Network.addListener('networkStatusChange', callback)
  }
}

export function haversineDistance(lat1, lon1, lat2, lon2) {
  const R = 6371e3
  const φ1 = lat1 * Math.PI / 180
  const φ2 = lat2 * Math.PI / 180
  const Δφ = (lat2 - lat1) * Math.PI / 180
  const Δλ = (lon2 - lon1) * Math.PI / 180

  const a = Math.sin(Δφ / 2) * Math.sin(Δφ / 2) +
            Math.cos(φ1) * Math.cos(φ2) *
            Math.sin(Δλ / 2) * Math.sin(Δλ / 2)
  const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a))

  return R * c
}

export function formatDistance(meters) {
  if (meters < 1000) return `${Math.round(meters)} m`
  return `${(meters / 1000).toFixed(1)} km`
}

export function getBearing(lat1, lon1, lat2, lon2) {
  const φ1 = lat1 * Math.PI / 180
  const φ2 = lat2 * Math.PI / 180
  const Δλ = (lon2 - lon1) * Math.PI / 180

  const y = Math.sin(Δλ) * Math.cos(φ2)
  const x = Math.cos(φ1) * Math.sin(φ2) - Math.sin(φ1) * Math.cos(φ2) * Math.cos(Δλ)
  const θ = Math.atan2(y, x)
  
  return (θ * 180 / Math.PI + 360) % 360
}

export function getCardinalDirection(bearing) {
  const directions = ['N', 'NE', 'E', 'SE', 'S', 'SW', 'W', 'NW']
  const index = Math.round(bearing / 45) % 8
  return directions[index]
}
